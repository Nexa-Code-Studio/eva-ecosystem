% --- Safe UI Figure Macro (with Placeholder Fallback) ---
% Usage: \uifigure[0.85\textwidth]{web/login.png}{Tampilan Login Portal}{fig:web-login}
\newcommand{\uifigure}[4][0.85\textwidth]{%
  \begin{figure}[H]
    \centering
    \IfFileExists{figures/#2}{%
      \includegraphics[width=#1]{figures/#2}%
    }{%
      \begin{tcolorbox}[
        width=#1,
        colback=EvaCanvas,
        colframe=EvaBorder,
        boxrule=1pt,
        arc=6pt,
        halign=center,
        valign=center,
        height=4.2cm
      ]
        {\bfseries\color{EvaPrimary} Placeholder Screenshot}\par\vspace{4pt}
        {\small\ttfamily\color{EvaMuted} figures/#2}
      \end{tcolorbox}
    }%
    \caption{#3}
    \label{#4}
  \end{figure}
}
